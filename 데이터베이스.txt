CREATE TABLE `emp_member` (
  `emp_id` char(12) NOT NULL DEFAULT '' COMMENT '아이디',
  `emp_pwd` char(15) DEFAULT NULL COMMENT '비밀번호',
  `emp_name` varchar(20) DEFAULT NULL COMMENT '이름',
  `emp_phone` char(11) DEFAULT NULL COMMENT '전화번호',
  `emp_email` varchar(30) DEFAULT NULL COMMENT '이메일',
  `emp_addr` varchar(30) DEFAULT NULL COMMENT '주소',
  `emp_reg_date` date DEFAULT NULL COMMENT '가입일',
  `emp_mod_date` date DEFAULT NULL COMMENT '수정일',
  PRIMARY KEY (`emp_id`)
);

CREATE TABLE emp_resume (
  emp_id char(12) NOT NULL DEFAULT '' COMMENT '아이디',
  emp_school    char(12)    comment "학교 및 상태",
  emp_midschool_name   varchar(12)   COMMENT '중학교 명',
  emp_midschool_grad   date   COMMENT '졸업년월',
  emp_highschool_name   varchar(12)   COMMENT '고등학교 명',
  emp_highschool_grad   date   COMMENT '졸업년월',
  emp_univer_name   varchar(12)   COMMENT '대학교 명',
  emp_univer_major   varchar(10)   COMMENT '전공',
  emp_univer_grad   date   COMMENT '졸업년월',
  emp_gradschool_name   varchar(12)   COMMENT '대학교 명',
  emp_gradschool_degree   char(10)   COMMENT '학위',
  emp_gradschool_major   varchar(10)   COMMENT '전공',
  emp_gradschool_grad   date   COMMENT '졸업년월',
  emp_Career    varchar(4)    comment '신입/경력',
  emp_company    varchar(20)    comment '회사명',
  emp_join_date   date   COMMENT '입사년월',
  emp_leave_date   date   COMMENT '입사년월',
  emp_employed   char(6)   COMMENT '재직여부',
  emp_job_description   text(500)   COMMENT '재직여부',
  emp_location   varchar(40)   COMMENT '근무지',
  emp_worktype   char(8)   COMMENT '재택근무 여부',
  emp_job_search   char(8)   COMMENT '직종 검색',
  emp_jobtype    varchar(26)   COMMENT '근무형태',
  
  
);
